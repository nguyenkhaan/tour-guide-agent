//User account 
table users {
    id UUID PK // Mã định danh duy nhất của người dùng
    email VARCHAR(255) // Địa chỉ email dùng để đăng nhập
    password_hash VARCHAR(255) // Mật khẩu đã băm an toàn
    full_name VARCHAR(100) // Họ và tên hiển thị
    role VARCHAR(20) // Phân quyền vai trò người dùng
    is_active BOOLEAN // Trạng thái kích hoạt tài khoản
    is_banned BOOLEAN // Cờ khóa tài khoản
    ban_reason TEXT // Lý do khóa tài khoản
    banned_at TIMESTAMP // Thời điểm khóa tài khoản
    banned_by UUID // Admin thực hiện khóa
    created_at TIMESTAMP // Thời điểm tạo tài khoản
    updated_at TIMESTAMP // Thời điểm cập nhật gần nhất
}

// Bảng Bộ nhớ Tạm của Agent trong Phiên chạy
table agent_working_memories {
id UUID PK // Định danh bộ nhớ tạm
session_id VARCHAR(100) // Định danh phiên làm việc
trip_id UUID // Gắn với chuyến đi
agent_type VARCHAR(30) // Agent sở hữu bộ nhớ
memory_key VARCHAR(100) // Khóa định danh ngữ cảnh
memory_value JSONB // Giá trị dữ liệu ngữ cảnh
expires_at TIMESTAMP // Hạn lưu trữ
created_at TIMESTAMP // Thời điểm khởi tạo
}

// Bảng Cấu hình Tham số Toàn cục Hệ thống
table system_configs {
    config_key VARCHAR(100) PK // Khóa định danh tham số cấu hình
    config_value JSONB // Giá trị cấu hình linh hoạt
    description TEXT // Mô tả ý nghĩa tham số
    updated_by UUID // Admin cập nhật cấu hình gần nhất
    updated_at TIMESTAMP // Thời điểm cập nhật
}
